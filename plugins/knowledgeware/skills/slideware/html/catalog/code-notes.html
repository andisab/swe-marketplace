from agent import tool
 
@tool(
    name="search_tickets",
    description="Search open tickets by text. Use before
        creating a ticket, to avoid duplicates.",
)
def search_tickets(
    query: str, limit: int = 10,
) -> list[Ticket]:
    # read-only: no side effects, safe to retry
    return db.search(query)[:limit]
